  from pytest_bdd.plugin.cucumber_json_dispatcher import entrypoint
  from types import SimpleNamespace

  def test_dispatcher_no_warning_on_cli_override_ini():
      ini_cache = {"cucumber_json_path": "/ini/path"}
      config = SimpleNamespace(
          _inicache=ini_cache,
          option=SimpleNamespace(cucumber_js_json_path="/cli/path"),  # CLI is set
      )
      config.getini = lambda key: ini_cache.get(key, "")
      # no workerinput attr — not a worker

      with warnings.catch_warnings(record=True) as warning_list:
          warnings.simplefilter("always")
          entrypoint.pytest_configure(config)

      assert len(warning_list) == 0  # no warnings emitted
      assert ini_cache["cucumber_json_path"] == ""  # INI was zeroed (CLI won)
</action>
