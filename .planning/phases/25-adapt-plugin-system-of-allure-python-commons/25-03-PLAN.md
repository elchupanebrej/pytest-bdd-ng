# Phase 25 Plan 03: Hook-Based Interception Architecture

**Implement MessageDrivenListener, AllureCucumberApiHooks, CucumberEnvelopeAdapter, and EnvelopeStateCache following allure-pytest/allure-pytest-bdd patterns**

## Context

Phase 25 Plans 01 and 02 rewired the plugin for envelope-based ingestion and created verification tests. However, the current implementation still collects messages into `self.envelopes` and converts them at `pytest_sessionfinish` (batch mode). This plan implements the architecture revision specified in SPEC R4 and R9: a `MessageDrivenListener` that intercepts `pytest_bdd_message` calls in real-time and writes Allure results through `AllureLifecycle` incrementally.

**Reference implementations (D-27):**
- `allure-framework/allure-python/allure-pytest-bdd/src/pytest_bdd_listener.py` — listener pattern
- `allure-framework/allure-python/allure-pytest-bdd/src/allure_api_listener.py` — API hooks pattern
- `allure-framework/allure-python/allure-pytest/src/listener.py` — cache pattern (ItemCache)
- `allure-framework/allure-python/allure-pytest-bdd/src/plugin.py` — registration and cleanup patterns

## Goal

Replace the batch-collecting `AllureCucumberPlugin` with a message-driven architecture that processes Cucumber envelopes in real-time, writing Allure results incrementally during test execution.

## Success Criteria (what must be TRUE)

1. `MessageDrivenListener` class exists and receives `pytest_bdd_message` hook calls
2. Each envelope is deserialized, routed by `payload_kind`, and written through `AllureLifecycle` incrementally
3. `AllureCucumberApiHooks` class exists and is registered with `allure_commons.plugin_manager`
4. `CucumberEnvelopeAdapter` routes envelopes by `payload_kind` to appropriate lifecycle methods
5. `EnvelopeStateCache` tracks `test_case_started_id → Allure uuid` mappings
6. `entrypoint.py` follows `cleanup_factory` registration pattern from allure-pytest-bdd
7. Allure result files are written to output directory incrementally during test execution
8. Existing contract tests from Plan 01 continue to pass
9. No import of `allure-pytest` anywhere in the plugin

```python
"""CucumberEnvelopeAdapter — state machine for envelope routing."""

import logging
from attrs import define, field
from uuid import uuid4

if TYPE_CHECKING:
    from allure_commons.lifecycle import AllureLifecycle
    from pytest_bdd.model.execution_message_adapter import ExecutionProjection

logger = logging.getLogger(__name__)


@define(eq=False, hash=False)
class EnvelopeStateCache:
    """Track test_case_started_id → Allure uuid mappings.

    Inspired by allure-pytest's ItemCache pattern (listener.py:283).
    """
    _case_uuids: dict[str, str] = field(factory=dict)
    _step_uuids: dict[str, str] = field(factory=dict)

    def get_case_uuid(self, case_started_id: str) -> str | None:
        return self._case_uuids.get(case_started_id)

    def set_case_uuid(self, case_started_id: str, uuid: str) -> None:
        self._case_uuids[case_started_id] = uuid

    def get_step_uuid(self, step_started_id: str) -> str | None:
        return self._step_uuids.get(step_started_id)

    def set_step_uuid(self, step_started_id: str, uuid: str) -> None:
        self._step_uuids[step_started_id] = uuid

    def clear(self) -> None:
        self._case_uuids.clear()
        self._step_uuids.clear()


@define(eq=False, hash=False)
class CucumberEnvelopeAdapter:
    """State machine routing envelopes by payload_kind to AllureLifecycle."""

    lifecycle: AllureLifecycle = field()
    state_cache: EnvelopeStateCache = field(factory=EnvelopeStateCache)

    def route_envelope(self, projection: ExecutionProjection) -> None:
        """Route a single envelope to the appropriate lifecycle method."""
        handler = getattr(self, f"_handle_{projection.payload_kind}", None)
        if handler is not None:
            handler(projection)
        else:
            logger.debug("Unmapped payload_kind: %s", projection.payload_kind)

    def _handle_test_case_started(self, projection: ExecutionProjection) -> None:
        """Schedule a new test case in AllureLifecycle."""
        case_uuid = str(uuid4())
        self.state_cache.set_case_uuid(projection.test_case_started_id, case_uuid)
        with self.lifecycle.schedule_test_case(case_uuid) as test_result:
            test_result.name = projection.test_case_name
            # Set labels, links, parameters from projection

    def _handle_test_step_started(self, projection: ExecutionProjection) -> None:
        """Start a new step in the current test case."""
        case_uuid = self.state_cache.get_case_uuid(projection.test_case_started_id)
        step_uuid = str(uuid4())
        self.state_cache.set_step_uuid(projection.test_step_started_id, step_uuid)
        with self.lifecycle.start_step(parent_uuid=case_uuid, uuid=step_uuid) as step:
            step.name = projection.test_step_name

    def _handle_test_step_finished(self, projection: ExecutionProjection) -> None:
        """Stop the current step and set its status."""
        step_uuid = self.state_cache.get_step_uuid(projection.test_step_started_id)
        self.lifecycle.stop_step(step_uuid)

    def _handle_test_case_finished(self, projection: ExecutionProjection) -> None:
        """Write the completed test case to Allure output."""
        case_uuid = self.state_cache.get_case_uuid(projection.test_case_started_id)
        self.lifecycle.write_test_case(case_uuid)
```

```python
"""MessageDrivenListener — receives pytest_bdd_message, maps to lifecycle."""

import logging
from attrs import define, field
from allure_commons.lifecycle import AllureLifecycle
from allure_commons.logger import AllureFileLogger
from pytest_bdd.plugin.allure_cucumber.message_adapter import CucumberEnvelopeAdapter

if TYPE_CHECKING:
    from pathlib import Path
    from pytest_bdd.compatibility.pytest import Config
    from pytest_bdd.model.message_extension import EventEnvelope

logger = logging.getLogger(__name__)


@define(eq=False, hash=False)
class MessageDrivenListener:
    """Receive pytest_bdd_message events and map to AllureLifecycle.

    Follows allure-pytest-bdd's PytestBDDListener pattern (pytest_bdd_listener.py:16).
    """
    config: Config = field()
    output_dir: str = field()
    lifecycle: AllureLifecycle = field(factory=AllureLifecycle)
    adapter: CucumberEnvelopeAdapter = field(init=False)
    file_logger: AllureFileLogger | None = field(default=None, init=False)

    def __attrs_post_init__(self) -> None:
        self.adapter = CucumberEnvelopeAdapter(lifecycle=self.lifecycle)

    def start(self) -> None:
        """Initialize the file logger and register with allure_commons."""
        from pathlib import Path
        import os
        import allure_commons

        output_path = Path(os.path.expandvars(self.output_dir)).expanduser().resolve()
        output_path.mkdir(parents=True, exist_ok=True)

        self.file_logger = AllureFileLogger(str(output_path))
        allure_commons.plugin_manager.register(self.file_logger)

    def stop(self) -> None:
        """Unregister the file logger."""
        import allure_commons

        if self.file_logger is not None:
            allure_commons.plugin_manager.unregister(self.file_logger)
            self.file_logger = None

    def pytest_bdd_message(self, config: Config, message: EventEnvelope) -> None:  # noqa: ARG002
        """Process a single Cucumber Message envelope in real-time."""
        from pytest_bdd.model.execution_message_adapter import ExecutionMessageAdapter

        try:
            projection = ExecutionMessageAdapter.deserialize(message)
            self.adapter.route_envelope(projection)
        except Exception:
            logger.exception("Failed to process envelope: %s", message)
```

```python
"""AllureCucumberApiHooks — implements allure_commons hooks."""

import logging
from attrs import define, field
from allure_commons._hooks import hookimpl

if TYPE_CHECKING:
    from pytest_bdd.compatibility.pytest import Config
    from allure_commons.lifecycle import AllureLifecycle

logger = logging.getLogger(__name__)


@define(eq=False, hash=False)
class AllureCucumberApiHooks:
    """Implement allure_commons hooks for step and attach operations.

    Follows allure-pytest-bdd's AllurePytestBddApiHooks pattern (allure_api_listener.py:15).
    """
    config: Config = field()
    lifecycle: AllureLifecycle = field()

    @hookimpl
    def start_step(self, uuid: str, title: str, params: dict | None = None) -> None:
        """Start a new step in the current test case."""
        with self.lifecycle.start_step(uuid=uuid) as step:
            step.name = title

    @hookimpl
    def stop_step(self, uuid: str, exc_type: type | None, exc_val: Exception | None, exc_tb: object | None) -> None:
        """Stop the current step."""
        self.lifecycle.stop_step(uuid)

    @hookimpl
    def attach_data(self, body: str | bytes, name: str, attachment_type: str | None, extension: str | None) -> None:
        """Attach data to the current test case or step."""
        from uuid import uuid4
        attachment_uuid = str(uuid4())
        self.lifecycle.attach_data(
            uuid=attachment_uuid,
            body=body,
            name=name,
            attachment_type=attachment_type,
            extension=extension,
        )

    @hookimpl
    def attach_file(self, source: str, name: str, attachment_type: str | None, extension: str | None) -> None:
        """Attach a file to the current test case or step."""
        from uuid import uuid4
        attachment_uuid = str(uuid4())
        self.lifecycle.attach_file(
            uuid=attachment_uuid,
            source=source,
            name=name,
            attachment_type=attachment_type,
            extension=extension,
        )
```

**Implementation details:**

```python
"""Provide entrypoint helpers."""

from __future__ import annotations

import logging
from typing import TYPE_CHECKING

if TYPE_CHECKING:
    from pytest_bdd.compatibility.pytest import Config, Parser

logger = logging.getLogger(__name__)


def pytest_addoption(parser: Parser) -> None:
    """Add pytest-bdd options."""
    group = parser.getgroup("bdd", "Allure Cucumber")
    group.addoption(
        "--allure-cucumber-output",
        action="store",
        dest="allure_cucumber_output_dir",
        metavar="PATH",
        default=None,
        help="Output directory for Allure result JSON files.",
    )
    group.addoption(
        "--allure-cucumber-messages-in",
        action="store",
        dest="allure_cucumber_messages_in",
        metavar="PATH",
        default=None,
        help="Path to an existing Cucumber Messages NDJSON file to import into Allure results.",
    )
    parser.addini(
        "allure_cucumber_output_dir",
        help="Output directory for Allure result JSON files.",
        default="allure-results",
    )


def _cleanup_listener(config: Config, listener: object) -> None:
    """Cleanup callback for listener unregister."""
    config.pluginmanager.unregister(listener)
    if hasattr(listener, 'stop'):
        listener.stop()


def pytest_configure(config: Config) -> None:
    """Handle configure."""
    import pytest
    import os
    from pathlib import Path

    output_dir = getattr(config.option, "allure_cucumber_output_dir", None) or config.getini(
        "allure_cucumber_output_dir",
    )
    messages_in = getattr(config.option, "allure_cucumber_messages_in", None)

    # Fail early if there is a configuration conflict
    if messages_in is not None and not output_dir:
        raise pytest.UsageError(
            "--allure-cucumber-messages-in requires --allure-cucumber-output to write results."
        )

    has_output = getattr(config.option, "allure_cucumber_output_dir", None) is not None
    has_input = messages_in is not None

    if not has_output and not has_input:
        return

    # Import and create listener
    from pytest_bdd.plugin.allure_cucumber.listener import MessageDrivenListener
    from pytest_bdd.plugin.allure_cucumber.api_hooks import AllureCucumberApiHooks
    import allure_commons

    output_path = Path(os.path.expandvars(output_dir or "allure-results")).expanduser().resolve()

    # Create and register listener
    listener = MessageDrivenListener(
        config=config,
        output_dir=str(output_path),
    )
    config.pluginmanager.register(listener, "allure-cucumber-listener")
    config.add_cleanup(_cleanup_listener, config, listener)

    # Register API hooks with allure_commons
    api_hooks = AllureCucumberApiHooks(
        config=config,
        lifecycle=listener.lifecycle,
    )
    allure_commons.plugin_manager.register(api_hooks)

    # Start the listener (registers file logger)
    listener.start()

    # Store references for unconfigure
    config._allure_cucumber_listener = listener  # type: ignore[attr-defined]  # noqa: SLF001
    config._allure_cucumber_api_hooks = api_hooks  # type: ignore[attr-defined]  # noqa: SLF001


def pytest_unconfigure(config: Config) -> None:
    """Handle unconfigure."""
    import allure_commons

    listener = getattr(config, "_allure_cucumber_listener", None)
    if listener is not None:
        del config._allure_cucumber_listener  # type: ignore[attr-defined]  # noqa: SLF001
        config.pluginmanager.unregister(listener)
        listener.stop()

    api_hooks = getattr(config, "_allure_cucumber_api_hooks", None)
    if api_hooks is not None:
        del config._allure_cucumber_api_hooks  # type: ignore[attr-defined]  # noqa: SLF001
        allure_commons.plugin_manager.unregister(api_hooks)
```
